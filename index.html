function animateCounter(el, target) {
  const duration = 1400;
  let start = null;
  function step(ts) {
    if (!start) start = ts;
    const p = Math.min((ts-start)/duration, 1);
    el.textContent = Math.round(target * p);
    if (p < 1) requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
}
